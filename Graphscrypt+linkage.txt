private void SetupTempChart()
{
    TempChart.Series = new ISeries[]
    {
new LineSeries<double>
{
    Values = tempReadings,
    GeometrySize = 5,
    Fill = null
}
    };
}

private void StartTempTimer()
{
    if (tempTimer != null)
    {
        return;
    }

    tempTimer = new DispatcherTimer();
    tempTimer.Interval = TimeSpan.FromSeconds(5);
    tempTimer.Tick += TempTimer_Tick;
    tempTimer.Start();

    // add one reading immediately so the graph isnt empty on load
    TempTimer_Tick(null, null);
}

private void TempTimer_Tick(object sender, EventArgs e)
{
    double newTemp = Math.Round(20 + rng.NextDouble() * 10, 1); // random value between 20 and 30

    tempReadings.Add(newTemp);

    if (tempReadings.Count > 15)
    {
        tempReadings.RemoveAt(0);
    }
    

    CurrentTempText.Text = "Current Temp: " + newTemp + " °C";
}

private void StopTempTimer()
{
    if (tempTimer != null)
    {
        tempTimer.Stop();
        tempTimer = null;
    }
}
